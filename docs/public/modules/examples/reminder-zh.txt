#!name=示例：定时提醒
#!desc=按设置的时间发一条通知，并在面板里显示运行了几次。不需要 HTTPS 解密。
#!arguments=时间:0 9 * * *
#!arguments-desc=时间按“分 时 日 月 星期”写，默认每天 9:00。想马上看到效果，可以先改成 * * * * *（每分钟一次），试完再改回来。

[Script]
定时提醒 = type=cron, cronexp="{{{时间}}}", timeout=10, script-path=https://clash.md/modules/examples/reminder-zh.js
提醒面板 = type=generic, timeout=10, script-path=https://clash.md/modules/examples/reminder-panel-zh.js

[Panel]
提醒面板 = title=定时提醒, content=还没有运行过, script-name=提醒面板, update-interval=3600
